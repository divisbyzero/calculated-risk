\begin{category}{Math Notation}

  % Shows: inline math with $...$, and the [say] option.
  \begin{clue}[say]
    It is the value of $2^{10}.$
  \response
    What is $1024$?
  \end{clue}

  % Shows: a macro defined at the top of the file (\dd), inline math
  % with \( ... \), and a one-line solution.
  \begin{clue}[say]
    It is the value of \(\dd{x}\left(x^3\right)\) at \(x = 2.\)
  \response
    What is $12$?
  \solution
    $\dd{x}\left(x^3\right) = 3x^2,$ and $3(2)^2 = 12.$
  \end{clue}

  % Shows: display math with \[ ... \], and a worked solution in align*.
  \begin{clue}[say]
    It is the value of
    \[ \int_0^{\pi} \sin x \, dx. \]
  \response
    What is $2$?
  \solution
    \begin{align*}
      \int_0^{\pi} \sin x \, dx &= \Big[-\cos x\Big]_0^{\pi} \\
        &= -\cos\pi + \cos 0 \\
        &= 2.
    \end{align*}
  \end{clue}

  % Shows: a piecewise function with the cases environment.
  \begin{clue}[say]
    It is $f(-3)$ for
    \[ f(x) = \begin{cases} x^2 & \text{if } x \ge 0, \\ -x & \text{if } x < 0. \end{cases} \]
  \response
    What is $3$?
  \end{clue}

  % Shows: a matrix.
  \begin{clue}[say]
    It is the determinant of
    \[ A = \begin{pmatrix} 1 & 2 \\ 3 & 4 \end{pmatrix}. \]
  \response
    What is $-2$?
  \solution
    $\det A = (1)(4) - (2)(3) = -2.$
  \end{clue}

\end{category}

\begin{category}{Pictures and Layout}

  % Shows: a picture in a clue. On the board it goes to the right of the
  % text; in the PDF, center puts it on its own line and [height=3.5cm]
  % sets its size.
  \begin{clue}[say]
    It is the area of the shaded region under $y = x^2.$
    \begin{center}
      \includegraphics[height=3.5cm]{shaded-area.png}
    \end{center}
  \response
    What is $9$?
  \solution
    $\displaystyle\int_0^3 x^2 \, dx = \frac{x^3}{3}\Big|_0^3 = 9.$
  \end{clue}

  % Shows: a picture in a solution.
  \begin{clue}[say]
    It is the hypotenuse of a right triangle with legs $5$ and $12.$
  \response
    What is $13$?
  \solution
    $\sqrt{5^2 + 12^2} = \sqrt{169} = 13.$
    \begin{center}
      \includegraphics[height=2cm]{right-triangle.png}
    \end{center}
  \end{clue}

  % Shows: line breaks with \\.
  \begin{clue}[say]
    Roses are red,\\
    violets are blue,\\
    it's the derivative of $x^2,$\\
    and it starts with a $2.$
  \response
    What is $2x$?
  \end{clue}

  % Shows: a long clue. The board shrinks text to fit the screen.
  \begin{clue}[say]
    It is the theorem that says: if $f$ is continuous on $[a,b]$ and
    differentiable on $(a,b),$ then there is some $c$ in $(a,b)$ where the
    tangent line is parallel to the secant line through the endpoints, that
    is,
    \[ f'(c) = \frac{f(b) - f(a)}{b - a}. \]
  \response
    What is the Mean Value Theorem?
  \end{clue}

  % Shows: an align* environment in a clue.
  \begin{clue}[say]
    It is the value of $x + y$ if
    \begin{align*}
      2x + y &= 7, \\
      x - y &= 2.
    \end{align*}
  \response
    What is $4$?
  \solution
    Adding the equations gives $3x = 9,$ so $x = 3$ and $y = 1.$
  \end{clue}

\end{category}

  \response

  \begin{clue}[write]
    It is the derivative of $\sin\left(x^2\right).$
  \response
    What is $2x\cos\left(x^2\right)$?
  \end{clue}

  % Shows: a clue with neither [write] nor [say]. The board shows no banner.
  \begin{clue}
    It is the next number in $1, 1, 2, 3, 5, 8, \ldots$
  \response
    What is $13$?
  \end{clue}

  % Shows: two options at once. This clue is a Double Down.
  \begin{clue}[doubledown, write]
    It is the tangent line to $y = x^2$ at $x = 3.$
  \response
    What is $y = 6x - 9$?
  \solution
    The slope is $y'(3) = 6$ and the point is $(3, 9),$ so
    $y - 9 = 6(x - 3),$ or $y = 6x - 9.$
  \end{clue}

  % Shows: a "What are" response with more than one answer.
  \begin{clue}[say]
    They are the solutions of $x^2 - 5x + 6 = 0.$
  \response
    What are $x = 2$ and $x = 3$?
  \solution
    $x^2 - 5x + 6 = (x - 2)(x - 3).$
  \end{clue}

\end{category}

% Shows: math in a category name.
\begin{category}{Sums $\Sigma$}

  \begin{clue}[say]
    It is the value of $\displaystyle\sum_{k=1}^{4} k.$
  \response
    What is $10$?
  \end{clue}

  \begin{clue}[say]
    It is the value of $\displaystyle\sum_{k=0}^{\infty} \left(\frac{1}{2}\right)^k.$
  \response
    What is $2$?
  \solution
    It is a geometric series with ratio $\frac{1}{2},$ so its sum is
    $\dfrac{1}{1 - \frac{1}{2}} = 2.$
  \end{clue}

  \begin{clue}[write]
    It is a formula for $\displaystyle\sum_{k=1}^{n} k.$
  \response
    What is $\dfrac{n(n+1)}{2}$?
  \end{clue}

  \begin{clue}[say]
    It is the word for what the harmonic series
    $\displaystyle\sum_{n=1}^{\infty} \frac{1}{n}$ does.
  \response
    What is diverge?
  \end{clue}

  \begin{clue}[say]
    It is the value of $\displaystyle\sum_{n=1}^{\infty} \frac{1}{n^2}.$
  \response
    What is $\dfrac{\pi^2}{6}$?
  \end{clue}

\end{category}

% Fermat's Last Round: one category and clue, after the categories.
\begin{final}{Famous Constants}
  It is the value of $\displaystyle\sum_{n=0}^{\infty} \frac{1}{n!}.$
\response
  What is $e$?
\solution
  The Taylor series of $e^x$ is $\displaystyle\sum_{n=0}^{\infty} \frac{x^n}{n!}.$
  Setting $x = 1$ gives $e.$
\end{final}
